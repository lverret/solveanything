# Equations
u(0, y) = 0
u(1, y) = y
u(x, 0) = 0
u(x, 1) = x**2
v(0, y) = 0
v(1, y) = -y**2
v(x, 0) = 0
v(x, 1) = -x
p(0, 0) = 0
u * grad(u, x) + v * grad(u, y) + grad(p, x) - 0.1 * (grad(grad(u, x), x) + grad(grad(u, y), y)) = x**3 * y**2 + 1 - 0.2 * y
u * grad(v, x) + v * grad(v, y) + grad(p, y) - 0.1 * (grad(grad(v, x), x) + grad(grad(v, y), y)) = x**2 * y**3 + 1 + 0.2 * x
grad(u, x) + grad(v, y) = 0

# Solution
u = x**2 * y
v = -x * y**2
p = x + y
